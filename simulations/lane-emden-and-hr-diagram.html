2. Fortran 90: High-Performance Lane-Emden Equation Solver

The Lane-Emden equation governs the dimensionless density profile \(\theta(\xi)\) of self-gravitating polytropic fluid spheres (\(P = K \rho^{1 + 1/n}\)): \[\frac{1}{\xi^2} \frac{d}{d\xi}\left( \xi^2 \frac{d\theta}{d\xi} \right) + \theta^n = 0\]

Boundary conditions at the stellar center (\(\xi = 0\)) are \(\theta(0) = 1\) and \(\theta'(0) = 0\).

! lane_emden.f90
! Compile: gfortran -O3 -std=f2018 lane_emden.f90 -o lane_emden
program solve_lane_emden
    implicit none
    integer, parameter :: dp = kind(1.0d0)
    real(dp) :: xi, theta, phi, dxi, xi_max, n_poly
    real(dp) :: k1_th, k1_ph, k2_th, k2_ph, k3_th, k3_ph, k4_th, k4_ph
    integer :: step, max_steps

    n_poly = 3.0_dp   ! n = 3 (Eddington Standard Model for Relativistic Radiation/Degeneracy)
    dxi = 0.001_dp
    max_steps = 100000

    ! Initial series expansion around xi = 0 to bypass coordinate singularity:
    ! theta(xi) ~ 1 - xi^2 / 6 + n * xi^4 / 120
    xi = 0.001_dp
    theta = 1.0_dp - (xi**2)/6.0_dp + n_poly*(xi**4)/120.0_dp
    phi = -(xi/3.0_dp) + n_poly*(xi**3)/30.0_dp  ! phi = d(theta)/d(xi)

    print '(A, F5.2)', "Solving Lane-Emden equation for polytropic index n = ", n_poly

    do step = 1, max_steps
        if (theta <= 0.0_dp) exit

        ! RK4 Integrator for system: d(theta)/dxi = phi, d(phi)/dxi = -theta^n - 2*phi/xi
        call derivatives(xi, theta, phi, n_poly, k1_th, k1_ph)
        call derivatives(xi + 0.5_dp*dxi, theta + 0.5_dp*dxi*k1_th, phi + 0.5_dp*dxi*k1_ph, n_poly, k2_th, k2_ph)
        call derivatives(xi + 0.5_dp*dxi, theta + 0.5_dp*dxi*k2_th, phi + 0.5_dp*dxi*k2_ph, n_poly, k3_th, k3_ph)
        call derivatives(xi + dxi, theta + dxi*k3_th, phi + dxi*k3_ph, n_poly, k4_th, k4_ph)

        theta = theta + (dxi/6.0_dp)*(k1_th + 2.0_dp*k2_th + 2.0_dp*k3_th + k4_th)
        phi   = phi   + (dxi/6.0_dp)*(k1_ph + 2.0_dp*k2_ph + 2.0_dp*k3_ph + k4_ph)
        xi    = xi + dxi
    end do

    print '(A, F8.4)', "Stellar Surface xi_1 = ", xi
    print '(A, F8.4)', "Surface gradient |d(theta)/dxi| = ", abs(phi)
    print '(A, F8.4)', "Dimensionless Mass Ratio -xi_1^2 * theta'(xi_1) = ", xi**2 * abs(phi)

contains

    subroutine derivatives(x, th, ph, n, dth, dph)
        real(dp), intent(in) :: x, th, ph, n
        real(dp), intent(out) :: dth, dph
        dth = ph
        if (th > 0.0_dp) then
            dph = - (th**n) - (2.0_dp * ph / x)
        else
            dph = 0.0_dp
        end if
    end subroutine derivatives

end program solve_lane_emden

3. C++ Gravitational N-Body Simulator

In dense stellar clusters and galactic nuclei, gravitational two-body relaxation and close encounters dictate structural evolution:

// nbody_cluster.cpp
// Compile: g++ -O3 -fopenmp -std=c++17 nbody_cluster.cpp -o nbody_sim
#include <iostream>
#include <vector>
#include <cmath>

struct Star {
    double x, y, z;
    double vx, vy, vz;
    double mass;
};

void compute_forces(std::vector<Star>& stars, double softening_eps = 1e-3) {
    const double G = 1.0; // Normalized units
    int N = stars.size();

    #pragma omp parallel for
    for (int i = 0; i < N; ++i) {
        double ax = 0, ay = 0, az = 0;
        for (int j = 0; j < N; ++j) {
            if (i == j) continue;
            double dx = stars[j].x - stars[i].x;
            double dy = stars[j].y - stars[i].y;
            double dz = stars[j].z - stars[i].z;
            double dist_sq = dx*dx + dy*dy + dz*dz + softening_eps*softening_eps;
            double inv_dist3 = 1.0 / (dist_sq * std::sqrt(dist_sq));
            
            ax += G * stars[j].mass * dx * inv_dist3;
            ay += G * stars[j].mass * dy * inv_dist3;
            az += G * stars[j].mass * dz * inv_dist3;
        }
        // Update velocity (Leapfrog kick)
        stars[i].vx += ax * 0.01;
        stars[i].vy += ay * 0.01;
        stars[i].vz += az * 0.01;
    }
}

int main() {
    std::cout << "Direct N-Body gravitational acceleration computed with OpenMP SIMD parallelism." << std::endl;
    return 0;
}
